@org.junit.Test
public void testGetStorageAttributeIntegerValueByName_11()
{
    final String mutationTriggerProperty = "agone.mutation.trigger";
    String originalValue = System.getProperty(mutationTriggerProperty);

    try
    {
        System.clearProperty(mutationTriggerProperty);

        try
        {
            new org.finra.herd.service.helper.StorageHelper().getStorageAttributeIntegerValueByName("attributeName",
                new org.finra.herd.model.jpa.StorageEntity(), false, false);
            org.junit.Assert.fail("Expected AssertionError to be thrown.");
        }
        catch (AssertionError e)
        {
            org.junit.Assert.assertEquals("AGONE_MUTATION_TRIGGER", e.getMessage());
        }
    }
    finally
    {
        if (originalValue == null)
        {
            System.clearProperty(mutationTriggerProperty);
        }
        else
        {
            System.setProperty(mutationTriggerProperty, originalValue);
        }
    }
}
