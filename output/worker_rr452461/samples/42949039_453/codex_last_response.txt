@Test
public void testGetStorageAttributeIntegerValueByName_12()
{
    String originalAgoneMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");
    try
    {
        String attributeName = "attributeName";
        boolean attributeRequired = true;
        boolean attributeValueRequiredIfExists = true;

        org.finra.herd.model.jpa.StorageAttributeEntity storageAttributeEntity = new org.finra.herd.model.jpa.StorageAttributeEntity();
        storageAttributeEntity.setName(attributeName);
        storageAttributeEntity.setValue("123");

        org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
        storageEntity.setName("storageName");
        storageEntity.getAttributes().add(storageAttributeEntity);

        Integer value = storageHelper.getStorageAttributeIntegerValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);

        org.junit.Assert.assertEquals(Integer.valueOf(123), value);
    }
    finally
    {
        if (originalAgoneMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalAgoneMutationTrigger);
        }
    }
}
