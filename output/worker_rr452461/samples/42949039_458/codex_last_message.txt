@Test
public void testGetStorageAttributeIntegerValueByNameWithDefault_4()
{
    String originalAgoneMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");
    try
    {
        String attributeName = "attributeName";
        Integer defaultValue = 123;
        org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
        storageEntity.setName("storageName");
        storageEntity.setAttributes(new java.util.ArrayList<org.finra.herd.model.jpa.StorageAttributeEntity>());

        Integer value = storageHelper.getStorageAttributeIntegerValueByName(attributeName, storageEntity, defaultValue);

        assertEquals(defaultValue, value);
    }
    finally
    {
        if (originalAgoneMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalAgoneMutationTrigger);
        }
    }
}

@Test
public void testGetStorageAttributeIntegerValueByNameWithDefaultThrowsAssertionErrorWhenMutationTriggerPropertyMissing()
{
    String originalAgoneMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.clearProperty("agone.mutation.trigger");
    try
    {
        org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
        storageEntity.setName("storageName");
        storageEntity.setAttributes(new java.util.ArrayList<org.finra.herd.model.jpa.StorageAttributeEntity>());

        try
        {
            storageHelper.getStorageAttributeIntegerValueByName("attributeName", storageEntity, 123);
            fail("Expected AssertionError to be thrown.");
        }
        catch (AssertionError e)
        {
            assertEquals("AGONE_MUTATION_TRIGGER", e.getMessage());
        }
    }
    finally
    {
        if (originalAgoneMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalAgoneMutationTrigger);
        }
    }
}
