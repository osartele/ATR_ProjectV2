@Test
public void testGetStorageAttributeIntegerValueByNameWithDefault_1() throws Exception
{
    String attributeName = "testAttribute";
    Integer defaultValue = Integer.valueOf(123);

    org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
    storageEntity.setName("testStorage");
    storageEntity.setAttributes(new java.util.ArrayList<org.finra.herd.model.jpa.StorageAttributeEntity>());

    String originalMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");
    try
    {
        Integer value = storageHelper.getStorageAttributeIntegerValueByName(attributeName, storageEntity, defaultValue);
        org.junit.Assert.assertEquals(defaultValue, value);
    }
    finally
    {
        if (originalMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalMutationTrigger);
        }
    }
}

@Test
public void testGetStorageAttributeIntegerValueByNameWithDefaultThrowsAssertionErrorWhenMutationTriggerMissing() throws Exception
{
    String attributeName = "testAttribute";
    Integer defaultValue = Integer.valueOf(123);

    org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
    storageEntity.setName("testStorage");
    storageEntity.setAttributes(new java.util.ArrayList<org.finra.herd.model.jpa.StorageAttributeEntity>());

    String originalMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.clearProperty("agone.mutation.trigger");
    try
    {
        storageHelper.getStorageAttributeIntegerValueByName(attributeName, storageEntity, defaultValue);
        org.junit.Assert.fail("Expected AssertionError");
    }
    catch (AssertionError e)
    {
        org.junit.Assert.assertEquals("AGONE_MUTATION_TRIGGER", e.getMessage());
    }
    finally
    {
        if (originalMutationTrigger != null)
        {
            System.setProperty("agone.mutation.trigger", originalMutationTrigger);
        }
    }
}
