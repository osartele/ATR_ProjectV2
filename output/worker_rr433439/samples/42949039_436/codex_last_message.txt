@Test
public void testGetStorageFileEntityByStorageUnitEntityAndFilePath()
{
    String originalMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");

    try
    {
        org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
        storageEntity.setName("STORAGE_NAME");

        org.finra.herd.model.jpa.StorageUnitEntity storageUnitEntity = new org.finra.herd.model.jpa.StorageUnitEntity();
        storageUnitEntity.setStorage(storageEntity);
        storageUnitEntity.setDirectoryPath("test/path");

        String filePath = "test/path/testFile.txt";
        String filePathWithoutDirectory = "testFile.txt";
        org.finra.herd.model.api.xml.BusinessObjectDataKey businessObjectDataKey = new org.finra.herd.model.api.xml.BusinessObjectDataKey();

        org.finra.herd.model.jpa.StorageFileEntity storageFileEntity = new org.finra.herd.model.jpa.StorageFileEntity();

        org.mockito.Mockito.when(storageFileDao.getStorageFileByStorageUnitEntityAndFilePath(storageUnitEntity, filePath)).thenReturn(null);
        org.mockito.Mockito.when(storageFileDao.getStorageFileByStorageUnitEntityAndFilePath(storageUnitEntity, filePathWithoutDirectory))
            .thenReturn(storageFileEntity);

        org.finra.herd.model.jpa.StorageFileEntity result =
            storageFileDaoHelper.getStorageFileEntity(storageUnitEntity, filePath, businessObjectDataKey);

        org.junit.Assert.assertSame(storageFileEntity, result);
        org.mockito.Mockito.verify(storageFileDao).getStorageFileByStorageUnitEntityAndFilePath(storageUnitEntity, filePath);
        org.mockito.Mockito.verify(storageFileDao).getStorageFileByStorageUnitEntityAndFilePath(storageUnitEntity, filePathWithoutDirectory);
    }
    finally
    {
        if (originalMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalMutationTrigger);
        }
    }
}
