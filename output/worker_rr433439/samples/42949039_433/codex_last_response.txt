@Test
public void testCreateStorageFileEntitiesFromStorageFiles()
{
    String originalAgoneMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "enabled");
    try
    {
        org.finra.herd.model.jpa.StorageUnitEntity storageUnitEntity = new org.finra.herd.model.jpa.StorageUnitEntity();
        String directoryPath = "test-directory";

        org.finra.herd.model.api.xml.StorageFile storageFile1 = new org.finra.herd.model.api.xml.StorageFile();
        storageFile1.setFilePath(directoryPath + "/file1.txt");
        storageFile1.setFileSizeBytes(123L);
        storageFile1.setRowCount(10L);

        org.finra.herd.model.api.xml.StorageFile storageFile2 = new org.finra.herd.model.api.xml.StorageFile();
        storageFile2.setFilePath(directoryPath + org.finra.herd.model.jpa.StorageFileEntity.S3_EMPTY_PARTITION);
        storageFile2.setFileSizeBytes(456L);
        storageFile2.setRowCount(20L);

        java.util.List<org.finra.herd.model.api.xml.StorageFile> storageFiles = java.util.Arrays.asList(storageFile1, storageFile2);

        java.util.List<org.finra.herd.model.jpa.StorageFileEntity> result =
            storageFileDaoHelper.createStorageFileEntitiesFromStorageFiles(storageUnitEntity, storageFiles, directoryPath);

        org.junit.Assert.assertEquals(2, result.size());
        org.junit.Assert.assertSame(storageUnitEntity, result.get(0).getStorageUnit());
        org.junit.Assert.assertSame(storageUnitEntity, result.get(1).getStorageUnit());
        org.junit.Assert.assertEquals(
            org.finra.herd.core.HerdStringUtils.getMinimizedFilePath(storageFile1.getFilePath(), directoryPath), result.get(0).getPath());
        org.junit.Assert.assertEquals(org.finra.herd.model.jpa.StorageFileEntity.S3_EMPTY_PARTITION, result.get(1).getPath());
        org.junit.Assert.assertEquals(storageFile1.getFileSizeBytes(), result.get(0).getFileSizeBytes());
        org.junit.Assert.assertEquals(storageFile1.getRowCount(), result.get(0).getRowCount());
        org.junit.Assert.assertEquals(storageFile2.getFileSizeBytes(), result.get(1).getFileSizeBytes());
        org.junit.Assert.assertEquals(storageFile2.getRowCount(), result.get(1).getRowCount());

        org.mockito.Mockito.verify(storageFileDao).saveStorageFiles(result);
    }
    finally
    {
        if (originalAgoneMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalAgoneMutationTrigger);
        }
    }
}

    {

        try
