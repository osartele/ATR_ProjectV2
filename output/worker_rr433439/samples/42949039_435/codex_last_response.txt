@org.junit.Test
public void testGetStorageFileEntityWithNullStorageFileEntity()
{
    String filePath = "filePath";
    String originalMutationTrigger = System.getProperty("agone.mutation.trigger");

    try
    {
        System.setProperty("agone.mutation.trigger", "true");
        org.mockito.Mockito.when(storageFileDao.getStorageFileByStorageNameAndFilePath(STORAGE_NAME, filePath)).thenReturn(null);

        try
        {
            storageFileDaoHelper.getStorageFileEntity(STORAGE_NAME, filePath);
            org.junit.Assert.fail("Expected ObjectNotFoundException to be thrown.");
        }
        catch (org.finra.herd.model.ObjectNotFoundException e)
        {
            org.junit.Assert.assertEquals(String.format("Storage file \"%s\" doesn't exist in \"%s\" storage.", filePath, STORAGE_NAME), e.getMessage());
        }
    }
    finally
    {
        if (originalMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalMutationTrigger);
        }
    }
}
