@Test
public void testGetStorageFileEntity()
{
    String filePath = "path/to/storage/file";
    StorageFileEntity storageFileEntity = new StorageFileEntity();

    String originalAgoneMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");
    try
    {
        org.mockito.Mockito.when(storageFileDao.getStorageFileByStorageNameAndFilePath(STORAGE_NAME, filePath)).thenReturn(storageFileEntity);

        StorageFileEntity result = storageFileDaoHelper.getStorageFileEntity(STORAGE_NAME, filePath);

        org.junit.Assert.assertSame(storageFileEntity, result);
        org.mockito.Mockito.verify(storageFileDao).getStorageFileByStorageNameAndFilePath(STORAGE_NAME, filePath);
    }
    finally
    {
        if (originalAgoneMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalAgoneMutationTrigger);
        }
    }
}
