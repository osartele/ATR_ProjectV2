@Test
public void testModifyTagsInSearchIndexBlankMessage()
{
    java.util.List<org.finra.herd.model.jpa.TagEntity> tagEntityList = new java.util.ArrayList<>();
    org.finra.herd.model.jpa.TagEntity tagEntity = new org.finra.herd.model.jpa.TagEntity();
    tagEntity.setId(1L);
    tagEntityList.add(tagEntity);

    String modificationType = "UPDATE";
    org.finra.herd.model.dto.SearchIndexUpdateDto searchIndexUpdateDto = new org.finra.herd.model.dto.SearchIndexUpdateDto(
        org.finra.herd.model.dto.SearchIndexUpdateDto.MESSAGE_TYPE_BUSINESS_OBJECT_DEFINITION_UPDATE, java.util.Arrays.asList(1L), modificationType);

    org.mockito.Mockito.when(configurationHelper.getProperty(org.finra.herd.model.dto.ConfigurationValue.SEARCH_INDEX_UPDATE_JMS_LISTENER_ENABLED))
        .thenReturn(Boolean.TRUE.toString());
    org.mockito.Mockito.when(jsonHelper.objectToJson(searchIndexUpdateDto)).thenReturn("");

    searchIndexUpdateHelper.modifyTagsInSearchIndex(tagEntityList, modificationType);

    org.mockito.Mockito.verify(jsonHelper).objectToJson(searchIndexUpdateDto);
    org.mockito.Mockito.verify(configurationHelper).getProperty(org.finra.herd.model.dto.ConfigurationValue.SEARCH_INDEX_UPDATE_JMS_LISTENER_ENABLED);
    org.mockito.Mockito.verify(notificationMessagePublishingService, org.mockito.Mockito.never())
        .addNotificationMessageToDatabaseQueue(org.mockito.ArgumentMatchers.any(org.finra.herd.model.dto.NotificationMessage.class));
}
