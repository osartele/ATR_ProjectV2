@Test
public void testGetPropertyValidationThrowsWhenConfigurationValueIsNull()
{
    MockEnvironment environment = new MockEnvironment();

    try
    {
        ConfigurationHelper.getProperty(null, String.class, environment);
        fail("expected NullPointerException, but no exception was thrown");
    }
    catch (Exception e)
    {
        assertEquals("thrown exception type", NullPointerException.class, e.getClass());
    }
}
