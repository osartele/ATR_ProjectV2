@Test
public void testModifyTagInSearchIndexSearchIndexUpdateJmsListenerDisabled()
{
    org.finra.herd.model.jpa.TagEntity tagEntity = new org.finra.herd.model.jpa.TagEntity();
    tagEntity.setId(123L);

    org.mockito.Mockito.when(configurationHelper.getProperty(org.finra.herd.model.api.xml.ConfigurationValue.SEARCH_INDEX_UPDATE_JMS_LISTENER_ENABLED))
        .thenReturn(Boolean.FALSE.toString());

    searchIndexUpdateHelper.modifyTagInSearchIndex(tagEntity, "UPDATE");

    org.mockito.Mockito.verifyNoMoreInteractions(jmsMessageHelper);
}
