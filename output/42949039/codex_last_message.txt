@Test
public void testModifyTagsInSearchIndexBlankMessage() throws Exception
{
    Long tagId1 = 1L;
    Long tagId2 = 2L;
    String modificationType = "UPDATE";

    org.finra.herd.model.jpa.TagEntity tagEntity1 = org.mockito.Mockito.mock(org.finra.herd.model.jpa.TagEntity.class);
    org.finra.herd.model.jpa.TagEntity tagEntity2 = org.mockito.Mockito.mock(org.finra.herd.model.jpa.TagEntity.class);
    org.mockito.Mockito.when(tagEntity1.getId()).thenReturn(tagId1);
    org.mockito.Mockito.when(tagEntity2.getId()).thenReturn(tagId2);

    java.util.List<org.finra.herd.model.jpa.TagEntity> tagEntityList = java.util.Arrays.asList(tagEntity1, tagEntity2);

    org.finra.herd.model.dto.SearchIndexUpdateDto expectedSearchIndexUpdateDto =
        new org.finra.herd.model.dto.SearchIndexUpdateDto(
            org.finra.herd.model.dto.SearchIndexUpdateDto.MESSAGE_TYPE_BUSINESS_OBJECT_DEFINITION_UPDATE,
            java.util.Arrays.asList(tagId1, tagId2), modificationType);

    org.mockito.Mockito.when(jsonHelper.objectToJson(org.mockito.Mockito.any(org.finra.herd.model.dto.SearchIndexUpdateDto.class)))
        .thenReturn(org.apache.commons.lang3.StringUtils.EMPTY);

    searchIndexUpdateHelper.modifyTagsInSearchIndex(tagEntityList, modificationType);

    org.mockito.Mockito.verify(jsonHelper).objectToJson(org.mockito.Mockito.refEq(expectedSearchIndexUpdateDto));
    org.mockito.Mockito.verify(sqsDao, org.mockito.Mockito.never())
        .sendMessage(org.mockito.Mockito.anyString(), org.mockito.Mockito.anyString());
}
