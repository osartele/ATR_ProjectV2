@Test
public void testGetStorageAttributeIntegerValueByName_3()
{
    String attributeName = "testAttribute";
    String attributeValue = "123";
    boolean attributeRequired = false;
    boolean attributeValueRequiredIfExists = false;

    StorageHelper localStorageHelper = new StorageHelper();

    org.finra.herd.model.jpa.StorageAttributeEntity storageAttributeEntity = new org.finra.herd.model.jpa.StorageAttributeEntity();
    storageAttributeEntity.setName(attributeName);
    storageAttributeEntity.setValue(attributeValue);

    org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
    storageEntity.setName("testStorage");
    storageEntity.setAttributes(new java.util.ArrayList<org.finra.herd.model.jpa.StorageAttributeEntity>(java.util.Arrays.asList(storageAttributeEntity)));

    try
    {
        localStorageHelper.getStorageAttributeIntegerValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);
        org.junit.Assert.fail("Expected IllegalStateException to be thrown.");
    }
    catch (IllegalStateException e)
    {
        org.junit.Assert.assertEquals(
            "Storage attribute \"" + attributeName + "\" must be a valid integer. Actual value is \"" + attributeValue + "\"",
            e.getMessage());
    }
}
