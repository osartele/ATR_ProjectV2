@org.junit.Test
public void testGetStorageAttributeIntegerValueByName_6()
{
    String attributeName = "attributeName";
    boolean attributeRequired = false;
    boolean attributeValueRequiredIfExists = false;
    org.finra.herd.model.jpa.StorageEntity storageEntity = org.mockito.Mockito.mock(org.finra.herd.model.jpa.StorageEntity.class);
    org.finra.herd.service.helper.StorageHelper storageHelperSpy =
        org.mockito.Mockito.spy(new org.finra.herd.service.helper.StorageHelper());

    String numericValue = "100";
    org.mockito.Mockito.doReturn(numericValue).when(storageHelperSpy)
        .getStorageAttributeValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);

    try
    {
        storageHelperSpy.getStorageAttributeIntegerValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);
        org.junit.Assert.fail("Expected IllegalStateException when attribute value is numeric.");
    }
    catch (IllegalStateException e)
    {
        org.junit.Assert.assertEquals(
            "Storage attribute \"" + attributeName + "\" must be a valid integer. Actual value is \"" + numericValue + "\"", e.getMessage());
    }

    String nonNumericValue = "ABC";
    org.mockito.Mockito.doReturn(nonNumericValue).when(storageHelperSpy)
        .getStorageAttributeValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);

    try
    {
        storageHelperSpy.getStorageAttributeIntegerValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);
        org.junit.Assert.fail("Expected NumberFormatException when attribute value is non-numeric.");
    }
    catch (NumberFormatException e)
    {
        org.junit.Assert.assertTrue(e.getMessage().contains(nonNumericValue));
    }
}
