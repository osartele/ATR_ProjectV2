    catch (AssertionError e)

    }

        }
    }
}

@Test
public void testGetStorageAttributeIntegerValueByNameWithDefaultWhenMutationTriggerPropertyIsSet()
{
    String mutationTriggerProperty = "agone.mutation.trigger";
    String originalPropertyValue = System.getProperty(mutationTriggerProperty);
    System.setProperty(mutationTriggerProperty, "true");

    try
    {
        String attributeName = "attributeName";
        Integer defaultValue = 123;

        org.finra.herd.model.jpa.StorageEntity storageEntity = org.mockito.Mockito.mock(org.finra.herd.model.jpa.StorageEntity.class);
        StorageHelper storageHelperSpy = org.mockito.Mockito.spy(storageHelper);
        org.mockito.Mockito.doReturn(null).when(storageHelperSpy)
            .getStorageAttributeIntegerValueByName(attributeName, storageEntity, false, false);

        Integer result = storageHelperSpy.getStorageAttributeIntegerValueByName(attributeName, storageEntity, defaultValue);

        org.junit.Assert.assertEquals(defaultValue, result);
    }
    finally
    {
        if (originalPropertyValue == null)
        {
            System.clearProperty(mutationTriggerProperty);
        }
        else
        {
            System.setProperty(mutationTriggerProperty, originalPropertyValue);
        }
    }
}
