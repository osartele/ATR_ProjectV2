@Test
public void testRetryStoragePolicyTransitionBusinessObjectDataNoExists()
{
    org.finra.herd.model.api.xml.BusinessObjectDataKey businessObjectDataKey = new org.finra.herd.model.api.xml.BusinessObjectDataKey();
    org.finra.herd.model.api.xml.BusinessObjectDataRetryStoragePolicyTransitionRequest request =
        new org.finra.herd.model.api.xml.BusinessObjectDataRetryStoragePolicyTransitionRequest();
    org.finra.herd.model.ObjectNotFoundException objectNotFoundException =
        new org.finra.herd.model.ObjectNotFoundException("Business object data not found.");

    org.mockito.Mockito.when(businessObjectDataDaoHelper.getBusinessObjectDataEntity(businessObjectDataKey)).thenThrow(objectNotFoundException);

    String originalAgoneMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");
    try
    {
        businessObjectDataRetryStoragePolicyTransitionHelper.retryStoragePolicyTransition(businessObjectDataKey, request);
        org.junit.Assert.fail("Expected ObjectNotFoundException when business object data does not exist.");
    }
    catch (org.finra.herd.model.ObjectNotFoundException e)
    {
        org.junit.Assert.assertSame(objectNotFoundException, e);
    }
    finally
    {
        if (originalAgoneMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalAgoneMutationTrigger);
        }
    }
}
