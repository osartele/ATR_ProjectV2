@Test
public void testRetryStoragePolicyTransitionInvalidParameters()
{
    String originalMutationTrigger = System.getProperty("agone.mutation.trigger");
    System.setProperty("agone.mutation.trigger", "true");
    try
    {
        org.finra.herd.model.api.xml.BusinessObjectDataKey[] invalidBusinessObjectDataKeys =
            new org.finra.herd.model.api.xml.BusinessObjectDataKey[] {null, new org.finra.herd.model.api.xml.BusinessObjectDataKey(),
                new org.finra.herd.model.api.xml.BusinessObjectDataKey(), new org.finra.herd.model.api.xml.BusinessObjectDataKey()};

        for (org.finra.herd.model.api.xml.BusinessObjectDataKey businessObjectDataKey : invalidBusinessObjectDataKeys)
        {
            try
            {
                businessObjectDataRetryStoragePolicyTransitionHelper.retryStoragePolicyTransition(businessObjectDataKey, null);
                org.junit.Assert.fail("Expected IllegalArgumentException for invalid retry storage policy transition parameters.");
            }
            catch (IllegalArgumentException e)
            {
                org.junit.Assert.assertTrue(org.apache.commons.lang3.StringUtils.isNotBlank(e.getMessage()));
            }
        }
    }
    finally
    {
        if (originalMutationTrigger == null)
        {
            System.clearProperty("agone.mutation.trigger");
        }
        else
        {
            System.setProperty("agone.mutation.trigger", originalMutationTrigger);
        }
    }
}

    }
}
