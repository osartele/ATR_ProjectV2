@Test
public void testGetStorageAttributeIntegerValueByName_7()
{
    String attributeName = "testAttribute";
    boolean attributeRequired = false;
    boolean attributeValueRequiredIfExists = false;

    org.finra.herd.model.jpa.StorageAttributeEntity storageAttributeEntity = new org.finra.herd.model.jpa.StorageAttributeEntity();
    storageAttributeEntity.setName(attributeName);
    storageAttributeEntity.setValue("123");

    org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
    storageEntity.setName("testStorage");
    storageEntity.getAttributes().add(storageAttributeEntity);

    try
    {
        new org.finra.herd.service.helper.StorageHelper()
            .getStorageAttributeIntegerValueByName(attributeName, storageEntity, attributeRequired, attributeValueRequiredIfExists);
        org.junit.Assert.fail("Expected IllegalStateException to be thrown.");
    }
    catch (IllegalStateException e)
    {
        org.junit.Assert.assertEquals(
            "Storage attribute \"" + attributeName + "\" must be a valid integer. Actual value is \"123\"",
            e.getMessage());
    }
}
