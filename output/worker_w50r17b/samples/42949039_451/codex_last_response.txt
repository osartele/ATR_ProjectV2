@org.junit.Test
public void testGetStorageAttributeIntegerValueByName_10()
{
    final String mutationTriggerProperty = "agone.mutation.trigger";
    String originalMutationTriggerValue = System.getProperty(mutationTriggerProperty);

    org.finra.herd.service.helper.StorageHelper storageHelper = new org.finra.herd.service.helper.StorageHelper();
    org.finra.herd.model.jpa.StorageEntity storageEntity = new org.finra.herd.model.jpa.StorageEntity();
    storageEntity.setName("testStorage");

    org.finra.herd.model.jpa.StorageAttributeEntity storageAttributeEntity = new org.finra.herd.model.jpa.StorageAttributeEntity();
    storageAttributeEntity.setName("testAttribute");
    storageAttributeEntity.setValue("123");

    java.util.List<org.finra.herd.model.jpa.StorageAttributeEntity> attributes =
        new java.util.ArrayList<org.finra.herd.model.jpa.StorageAttributeEntity>();
    attributes.add(storageAttributeEntity);
    storageEntity.setAttributes(attributes);

    try
    {
        System.clearProperty(mutationTriggerProperty);
        try
        {
            storageHelper.getStorageAttributeIntegerValueByName("testAttribute", storageEntity, true, true);
            org.junit.Assert.fail("Expected AssertionError when mutation trigger property is not set.");
        }
        catch (AssertionError e)
        {
            org.junit.Assert.assertEquals("AGONE_MUTATION_TRIGGER", e.getMessage());
        }

        System.setProperty(mutationTriggerProperty, "true");
        Integer value = storageHelper.getStorageAttributeIntegerValueByName("testAttribute", storageEntity, true, true);
        org.junit.Assert.assertEquals(java.lang.Integer.valueOf(123), value);
    }
    finally
    {
        if (originalMutationTriggerValue == null)
        {
            System.clearProperty(mutationTriggerProperty);
        }
        else
        {
            System.setProperty(mutationTriggerProperty, originalMutationTriggerValue);
        }
    }
}
